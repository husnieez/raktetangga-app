            $keys = array_keys($avail);

class CheckoutModal extends Component
{
    public bool $isOpen = false;

    public string $borrowerUnit = 'Blok A-05 / RT 04';

        'open-checkout' => 'open',
    ];

    public function open(array $itemData): void
    {
        $this->item = $itemData;
        $this->isOpen = true;
        $this->stampedSuccess = false;
    }

    public function close(): void
    {
        $this->isOpen = false;
    }

    public function incrementDays(): void
    {
        if ($this->days < 14) {
            $this->days++;
        }
    }

    public function decrementDays(): void
    {
        if ($this->days > 1) {
            $this->days--;
        }
    }

    public function getRentalFeeProperty(): int
    {
        return ($this->item['daily_rate'] ?? 0) * $this->days;
    }

    public function getDepositFeeProperty(): int
    {
        return $this->item['deposit'] ?? 0;
    }

    public function getTotalPayProperty(): int
    {
        return $this->rentalFee + $this->depositFee;
    }

    public function getDueDateStringProperty(): string
    {
        $months = ['JAN', 'FEB', 'MAR', 'APR', 'MEI', 'JUN', 'JUL', 'AGU', 'SEP', 'OKT', 'NOV', 'DES'];
        $ts = strtotime("+{$this->days} days");
        $day = date('d', $ts);
        $month = $months[(int)date('n', $ts) - 1];
        $year = date('Y', $ts);
        return "{$day} {$month} {$year}";
    }

    public function confirmBooking(): void
    {
        $this->stampedSuccess = true;
        $this->dispatch('rental-confirmed', [
            'item_id' => $this->item['id'] ?? null,
            'days' => $this->days,
            'due_date' => $this->dueDateString,
            'location' => $this->handoverLocation,
        ]);
    }

    public function render()
    {
        return view('livewire.checkout-modal');
    }
}

            ];
        }
    }

    public function close(): void
    {
        $this->isOpen = false;
    }

    public function toggleComponent(int $id): void
    {
        foreach ($this->components as &$comp) {
            if ($comp['id'] === $id) {
                $comp['checked'] = !$comp['checked'];
                break;
            }
        }
    }

    public function getIsAllCheckedProperty(): bool
    {
        foreach ($this->components as $comp) {
            if (!$comp['checked']) return false;
        }
        return true;

    {
        if ($this->isAllChecked) return true;

        return false;

    {
        if (!$this->canSubmit) return;

        $this->dispatch('checklist-verified', [

            'note' => $this->missingNote,
        ]);
    }

    public function render()
    {
        return view('livewire.component-checklist-modal');
    }
